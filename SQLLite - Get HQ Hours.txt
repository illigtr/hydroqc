SELECT
	strftime('%Y-%m-%d', S.start_ts, 'unixepoch','localtime') AS date
	,sum(CASE WHEN strftime('%H', S.start_ts, 'unixepoch','localtime') = '00' THEN S.state END) AS '00'
	,sum(CASE WHEN strftime('%H', S.start_ts, 'unixepoch','localtime') = '01' THEN S.state END) AS '01'
	,sum(CASE WHEN strftime('%H', S.start_ts, 'unixepoch','localtime') = '02' THEN S.state END) AS '02'
	,sum(CASE WHEN strftime('%H', S.start_ts, 'unixepoch','localtime') = '03' THEN S.state END) AS '03'
	,sum(CASE WHEN strftime('%H', S.start_ts, 'unixepoch','localtime') = '04' THEN S.state END) AS '04'
	,sum(CASE WHEN strftime('%H', S.start_ts, 'unixepoch','localtime') = '05' THEN S.state END) AS '05'
	,sum(CASE WHEN strftime('%H', S.start_ts, 'unixepoch','localtime') = '06' THEN S.state END) AS '06'
	,sum(CASE WHEN strftime('%H', S.start_ts, 'unixepoch','localtime') = '07' THEN S.state END) AS '07'
	,sum(CASE WHEN strftime('%H', S.start_ts, 'unixepoch','localtime') = '08' THEN S.state END) AS '08'
	,sum(CASE WHEN strftime('%H', S.start_ts, 'unixepoch','localtime') = '09' THEN S.state END) AS '09'
	,sum(CASE WHEN strftime('%H', S.start_ts, 'unixepoch','localtime') = '10' THEN S.state END) AS '10'
	,sum(CASE WHEN strftime('%H', S.start_ts, 'unixepoch','localtime') = '11' THEN S.state END) AS '11'
	,sum(CASE WHEN strftime('%H', S.start_ts, 'unixepoch','localtime') = '12' THEN S.state END) AS '12'
	,sum(CASE WHEN strftime('%H', S.start_ts, 'unixepoch','localtime') = '13' THEN S.state END) AS '13'
	,sum(CASE WHEN strftime('%H', S.start_ts, 'unixepoch','localtime') = '14' THEN S.state END) AS '14'
	,sum(CASE WHEN strftime('%H', S.start_ts, 'unixepoch','localtime') = '15' THEN S.state END) AS '15'
	,sum(CASE WHEN strftime('%H', S.start_ts, 'unixepoch','localtime') = '16' THEN S.state END) AS '16'
	,sum(CASE WHEN strftime('%H', S.start_ts, 'unixepoch','localtime') = '17' THEN S.state END) AS '17'
	,sum(CASE WHEN strftime('%H', S.start_ts, 'unixepoch','localtime') = '18' THEN S.state END) AS '18'
	,sum(CASE WHEN strftime('%H', S.start_ts, 'unixepoch','localtime') = '19' THEN S.state END) AS '19'
	,sum(CASE WHEN strftime('%H', S.start_ts, 'unixepoch','localtime') = '20' THEN S.state END) AS '20'
	,sum(CASE WHEN strftime('%H', S.start_ts, 'unixepoch','localtime') = '21' THEN S.state END) AS '21'
	,sum(CASE WHEN strftime('%H', S.start_ts, 'unixepoch','localtime') = '22' THEN S.state END) AS '22'
	,sum(CASE WHEN strftime('%H', S.start_ts, 'unixepoch','localtime') = '23' THEN S.state END) AS '23'
FROM STATISTICS AS S
INNER JOIN statistics_meta AS M ON S.metadata_id = M.id
WHERE
M.statistic_id = 'sensor.hydroqc_maison_total_hourly_consumption'
AND date > '2025-11-14'
GROUP BY date
ORDER BY date DESC
